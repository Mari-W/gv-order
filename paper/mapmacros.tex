% Macros for the TeX that paper/tools/map2tex.py takes from the proof map.
% The page renderer tools/proof-map/page/index.html defines only
% \llbracket and \rrbracket beyond MathJax; stmaryrd gives both here.

\RequirePackage{amssymb}
\RequirePackage{stmaryrd}
\RequirePackage{adjustbox}
\RequirePackage{fancyvrb}
\RequirePackage{newunicodechar}
\RequirePackage{longtable}
\RequirePackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds}

% Lock-free additions (\inmode{lf} in the map). One scheme for the whole
% paper: a light shaded background behind the lock-free parts of a formula,
% with no change of text colour, so that it survives greyscale printing.
% In running text, where a box cannot break, the lock-free parts keep the
% accent colour of the shading instead.
\definecolor{lfbg}{HTML}{E6DDF0}
\definecolor{lfcolor}{HTML}{6A3D80}
\makeatletter
\newcommand{\lf@box}[2]{{\setlength{\fboxsep}{0.6pt}\colorbox{lfbg}{$\m@th#1#2$}}}
\newcommand{\lfadd}[1]{\ifmmode
    \mathchoice{\lf@box\displaystyle{#1}}{\lf@box\textstyle{#1}}%
               {\lf@box\scriptstyle{#1}}{\lf@box\scriptscriptstyle{#1}}%
  \else{\color{lfcolor}#1}\fi}
\makeatother
% A deadlock-free part that the lock-free instance replaces.
\newcommand{\lfalt}[2]{#1\,\lfadd{/\,#2}}

% Generated definitions, one per block, rule and table.
\makeatletter
\newcommand{\map@get}[2]{%
  \ifcsname map@#1@#2\endcsname
    \csname map@#1@#2\endcsname
  \else
    \PackageError{mapmacros}{No map #1 `#2'. Rerun paper/tools/map2tex.py}{}%
  \fi}
\newcommand{\mapblock}[1]{\map@get{block}{#1}}
\newcommand{\mapblockdf}[1]{\map@get{blockdf}{#1}}
\newcommand{\mapblocklf}[1]{\map@get{blocklf}{#1}}
\newcommand{\maprule}[1]{\map@get{rule}{#1}}
\newcommand{\mapruledf}[1]{\map@get{ruledf}{#1}}
\newcommand{\maptable}[1]{\map@get{table}{#1}}
\makeatother

% A formula scaled down to the line width and the page height when needed.
\newcommand{\mapfit}[1]{%
  \begin{adjustbox}{max width=\linewidth,max totalheight=0.8\textheight,center}%
    $\displaystyle #1$%
  \end{adjustbox}}
\newcommand{\mapfittext}[1]{%
  \begin{adjustbox}{max width=\linewidth,max totalheight=0.8\textheight,center}%
    #1%
  \end{adjustbox}}
% Rules side by side in one row of a figure.
\newcommand{\maprow}[1]{\mapfit{#1}\par\medskip}
\newcommand{\rulesep}{\qquad}

% Appendix layout. Each block of the map is a numbered item, so that the
% body can point at the formal definition of what it states in words.
\newcounter{mapblk}[section]
\renewcommand{\themapblk}{\thesection.\arabic{mapblk}}
\newcommand{\mapblockhead}[3]{%
  \par\medskip\noindent\refstepcounter{mapblk}%
  {\bfseries\themapblk\quad #1\ifx&#3&\else\ \lfadd{\textsf{(LF)}}\fi}\label{app:#2}%
  \hfill{\scriptsize\texttt{#2}}\par\nopagebreak\smallskip}
\newcommand{\mapexamplehead}[2]{\par\medskip\noindent{\itshape #1}\label{app:#2}\hfill{\scriptsize\texttt{#2}}\par\nopagebreak}
\newcommand{\mapdfvariant}{\par\noindent{\small Deadlock-free instance.}\par\nopagebreak}
\newcommand{\mapsource}[1]{\par\noindent{\scriptsize\raggedright Rocq: #1\par}\medskip}

% Example listings, generated into gen/examples/ with \lstkw and \lstcmt.
\definecolor{kwcolor}{HTML}{1F4E8C}
\definecolor{cmtcolor}{gray}{0.42}
\newcommand{\lstkw}[1]{{\color{kwcolor}#1}}
\newcommand{\lstcmt}[1]{{\color{cmtcolor}#1}}
\DefineVerbatimEnvironment{maplisting}{Verbatim}%
  {fontsize=\footnotesize,xleftmargin=0.5em,samepage=true,commandchars=\\\{\}}
