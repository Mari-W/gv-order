\section{Reductions for leader election example}
\label{sec:reduct-lead-elect}

As an example, we trace the reduction sequence starting from $p_0$  where $\fFunc{e_0}$
communicates first with $\fFunc{e_1}$
(\rocq{leader_proc}, \rocq{leader_sem_ptyped}).
\begin{itemize}
\item $\tau$ transition (internal communication) to
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tApp{\tClos}{(\tApp{\tSend}{\tPair{true}{d}})}}
      \\
      \fFunc{e_1} &=
                    \fFunc{\tLet[\tLVB][d]{\tPair{false}{d}}{\tLet*{\tApp{\tClos}{cs}}{\tIf{\tLVB}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{\tLet[\tLVB][d]{\tApp{\tRecv}{d}}{\tLet*{\tApp{\tClos}{cs}}{\tLet*{\tApp{\tClos}{d}}{\tIf{\tLVB}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transition ($\beta$ pair elimination) to
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tApp{\tClos}{(\tApp{\tSend}{\tPair{true}{d}})}}
      \\
      \fFunc{e_1} &=
                    \fFunc{{\tLet*{\tApp{\tClos}{cs}}{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{\tLet[\tLVB][d]{\tApp{\tRecv}{d}}{\tLet*{\tApp{\tClos}{cs}}{\tLet*{\tApp{\tClos}{d}}{\tIf{\tLVB}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transition (internal communication) to
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tApp{\tClos}{d}}
      \\
      \fFunc{e_1} &=
                    \fFunc{{\tLet*{\tApp{\tClos}{cs}}{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{\tLet[\tLVB][d]{\tPair{true}{d}}{\tLet*{\tApp{\tClos}{cs}}{\tLet*{\tApp{\tClos}{d}}{\tIf{\tLVB}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transition ($\beta$ pair elimination) to
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tApp{\tClos}{d}}
      \\
      \fFunc{e_1} &=
                    \fFunc{{\tLet*{\tApp{\tClos}{cs}}{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{\tLet*{\tApp{\tClos}{cs}}{\tLet*{\tApp{\tClos}{d}}{\tIf{true}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transition (internal communication, closing $\fFunc{cs}$,
  new $C = \set{\fFunc{c}, \fFunc{d}}$)
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tApp{\tClos}{d}}
      \\
      \fFunc{e_1} &=
                    \fFunc{{\tLet*{\tUnit}{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{\tLet*{\tUnit}{\tLet*{\tApp{\tClos}{d}}{\tIf{true}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transitions (2x $\beta$ unit)
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tApp{\tClos}{d}}
      \\
      \fFunc{e_1} &=
                    \fFunc{{{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{{\tLet*{\tApp{\tClos}{d}}{\tIf{true}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transition (internal communication, closing $\fFunc{d}$,
  new $C = \set{\fFunc{c}}$)
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tUnit}
      \\
      \fFunc{e_1} &=
                    \fFunc{{{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{{\tLet*{\tUnit}{\tIf{true}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transition (unit elim)
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tUnit}
      \\
      \fFunc{e_1} &=
                    \fFunc{{{\tIf{false}{\tApp{\tClos}{(select~left~c)}}{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{{{\tIf{true}{\tApp{\tClos}{(select~right~c)}}{\tUnit}}}}}
    \end{align*}
  \end{small}
\item $\tau$ transitions (2x $\beta$ bool)
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tUnit}
      \\
      \fFunc{e_1} &=
                    \fFunc{{{{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{{{{\tApp{\tClos}{(select~right~c)}}}}}}
    \end{align*}
  \end{small}
\item There are two remaining labeled transitions: $\lSend{c}{right}$
  followed by $\lTerm{c}$. The resulting process is terminated and is
  an element of $\lrProc{n}{\fType{\tEmpty} \vdash \fType{\tUnit}}$:
  \begin{small}
    \begin{align*}
      \fFunc{e_0} &=
                    \fFunc{\tUnit}
      \\
      \fFunc{e_1} &=
                    \fFunc{{{{\tUnit}}}}
      \\
      \fFunc{e_2} &=
                    \fFunc{{{{{\tUnit}}}}}
    \end{align*}
  \end{small}
\end{itemize}
