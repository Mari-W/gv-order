\begin{definition}[Algorithmic judgments \rocq{syn}]
\label{def:algo}
The algorithmic judgments have the forms
\[
  \cGamma ; \lenv \vdash^{F}_{W\lf{,N}} t \Rightarrow \ty \eff{\varepsilon}
  \qquad
  \cGamma ; \lenv \vdash^{F}_{W\lf{,N}} t \Leftarrow \ty \eff{\varepsilon}
\]
with $\cGamma$, $\lenv$, $F$, $W$ and $N$ as in \cref{def:judgment} and
\begin{center}
\small
\begin{tabular}{@{}l@{\quad}l@{}}
  $t$ & annotated term, $|t|$ its erasure to a term (\cref{app:alg-annot}) \\
  $\Rightarrow \ty$ & synthesis, $\ty$ and $\lf{\varepsilon}$ are outputs \\
  $\Leftarrow \ty$ & checking, $\ty$ is an input and $\lf{\varepsilon}$ an output
\end{tabular}
\end{center}
A binary rule gives the first premise the entries of $\lenv$ that its term mentions and the second premise the rest, both the entries of box type (\cref{app:alg-judgment}).
\end{definition}

\begin{definition}[Oracles]
\label{def:oracles}
The algorithm takes a decision procedure for each of
\begin{center}
\small
\begin{tabular}{@{}l@{\quad}l@{}}
  $\cGamma \vdash_W x \prec y$ & entailment of a fact \\
  $\cGamma \vdash_W C[\bar x]$ & an instance of a constraint, for E-Force (\cref{app:r-boxes}) \\
  $\conserv_{\cGamma}(\bar\iota, C)$ & conservativity, for E-Open (\cref{app:r-ex}) \\
  $\acyc_{\rho}(C)$, $\conserv_{\cGamma}(\rho, C)$, $\lf{\rankable_{\cGamma}(\rho, C)}$ & creation, for \ruleref{E-New} \\
  $C = C'$ & equality of constraints \\
  $\sty \approx \sty'$, $\lf{\sty \sqsubseteq \sty'}$ & equivalence and subtyping of session types (\cref{app:t-subs})
\end{tabular}
\end{center}
\end{definition}

\begin{theorem}[Soundness \rocq{algo_sound}]
\label{thm:algo-sound}
If $\cGamma ; \lenv \vdash^{F}_{W\lf{,N}} t \Rightarrow \ty \eff{\varepsilon}$ or $\cGamma ; \lenv \vdash^{F}_{W\lf{,N}} t \Leftarrow \ty \eff{\varepsilon}$, then $\cGamma ; \lenv \vdash^{F}_{W\lf{,N}} |t| : \ty \eff{\varepsilon}$.
\end{theorem}

\begin{theorem}[Completeness \rocq{algo_complete} \rocq{algo_iff}]
\label{thm:algo-complete}
Let $\lenv$ hold only contractive session types. If $\cGamma ; \lenv \vdash^{F}_{W\lf{,N}} \tm : \ty \eff{\varepsilon}$, then $\cGamma ; \lenv \vdash^{F}_{W\lf{,N}} t \Rightarrow \ty \eff{\varepsilon}$ for some $t$ with $|t| = \tm$.
\end{theorem}

\begin{theorem}[Decidability relative to the oracles \rocq{syn_dec} \rocq{chk_dec} \rocq{alg_sound} \rocq{alg_complete}]
\label{thm:algo-dec}
For the oracles of \cref{def:oracles}, a function $\mathsf{alg}$ computes, by recursion on $t$,
\[
  \mathsf{alg}(t)(\cGamma, \lenv, F) = (\ty\lf{, \varepsilon}) \iff \cGamma ; \lenv \vdash^{F}_{W\lf{,N}} t \Rightarrow \ty \eff{\varepsilon}
\]
and the same for checking. So both judgments are decidable and have at most one result.
\end{theorem}
