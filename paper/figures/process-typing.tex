% Process typing. From map block p-typing. P-Connect^n is in the appendix.
% Bookkeeping premises are stated in words (PLAN.md, presentation).
\begin{figure}[tb]
  \small
  \judghead{Typing processes}{\Sigma \vdash_{W}^{\lf{N,\eta}} p : T}
  \begin{rules}
    \infer*[lab=P-Term]
      {\tfree\,T \\ \pfree\,\Sigma \\ \lfalt{\cons}{\ranked}\,W \\ \pairs^{\lf{N}}(\Theta_B, C_0) \\\\
       \lf{\forall o \in \mathrm{dom}\,\Sigma.\ \tk_\eta(o) \leq N} \\
       \tyj{\varnothing}{\Sigma\lf{^{\eta}}, \Theta_B}{\varnothing}{t}{T}\eff{\varepsilon}}
      {\Sigma \vdash_{W}^{\lf{N,\eta}} (C_0, \{t\}) : T}
    \label{rule:P-Term}
    \and
    \infer*[lab=P-Connect]
      {\Sigma_1 \cup \Sigma_C \vdash_{W_1}^{\lf{N,\eta}} p_1 : T \\
       \Sigma_2 \cup \dual{\Sigma_C} \vdash_{W_2}^{\lf{N,\eta}} p_2 : \ast \\\\
       \lf{\forall o \in \mathrm{dom}\,\Sigma_C.\ \tk_{\eta}(o) + \tk_{\eta}(\dual{o}) \leq N} \\
       \lfalt{\cons}{\ranked}(W_1 \wjoin W_2)}
      {\Sigma_1 \cup \Sigma_2 \vdash_{W_1 \wjoin W_2}^{\lf{N,\eta}} p_1 \cpar{\Sigma_C} p_2 : T}
    \label{rule:P-Connect}
  \end{rules}
  \figrule
  \Description{Inference rules P-Term and P-Connect.}
  \caption{Process typing, under \cref{conv:lf,conv:implicit}. $\tfree$ is defined in \cref{def:side}, $\pfree$, $\pairs$, $\dual{\Sigma_C}$ and $\cpar{\Sigma_C}$ in \cref{def:proc-judgment}, and the $n$-ary rule P-Connect$^n$ is in \cref{app:p-typing}.}
  \label{fig:process-typing}
\end{figure}
