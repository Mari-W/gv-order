% Aging. From map block t-agecap. The structural cases are one line.
\begin{figure}[tb]
  \small
  \judghead{Aging}{T - c = T'}
  \begin{rules}
    \infer*[lab=Age-Credit]{c \leq a}{S_x^{a,m} - c = S_x^{a - c, m}}
    \label{rule:Age-Credit}
    \and
    \infer*[lab=Age-Idle]{\unfold(S) = \Skip}{S_x^{a,m} - c = S_x^{\max(a - c, 0), m}}
    \label{rule:Age-Idle}
    \and
    \infer*[lab=Age-Arr]{c \leq a}{(T \multimap U)^{a} - c = (T \multimap U)^{a - c}}
    \label{rule:Age-Arr}
    \and
    \infer*[lab=Age-Free]{\text{$T$ holds no end}}{T - c = T}
    \label{rule:Age-Free}
  \end{rules}
  \[ (T_1 \otimes T_2) - c = (T_1 - c) \otimes (T_2 - c) \quad\ \ \text{and likewise for $\oplus$ and $\exists$} \]
  \figrule
  \Description{Inference rules for aging a type by a cost.}
  \caption{Aging (lock-free only)}
  \label{fig:aging}
\end{figure}
