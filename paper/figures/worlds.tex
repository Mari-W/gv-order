% Points, moves, worlds. From map blocks syn-labels, syn-points, syn-moves,
% w-world, w-order, w-wf, w-ext. Restriction and the join lemma (w-ext) are
% in the appendix. Point constants are written rho (map: a).
\begin{figure}[tb]
  \small
  \judghead{Points}{x}
  \[
  \begin{array}{@{}c@{}}
    b ::= \rho \mid \iota \qquad
    x, y, z ::= b \mid \pplus{x} \mid \fld{x}{\kappa} \mid \dual{x} \qquad (\kappa \in \mathbb{N})
    \qquad\quad
    \dual{\dual{x}} = x \qquad \fld{\dual{x}}{\kappa} = \fld{x}{\kappa} \qquad \dual{\pplus{x}} = \pplus{\dual{x}}
  \end{array}
  \]
  \judghead{Where an end moves, where a payload arrives}{x \mapsto x'}
  \[
    x \mapsto \pplus{x} \qquad
    \mathsf{sel}(x, i) = \dual{\fld{x}{i}} \qquad
    \mathsf{bra}(x, i) = \fld{x}{i} \qquad
    \mathsf{payload}(x, n) = (\fld{x}{0}, \ldots, \fld{x}{(n{-}1)})
  \]
  \judghead{Worlds}{W}
  \[
  \begin{array}{@{}r@{\ \ }c@{\ \ }l@{}}
    W & = & (A, {\prec_A}) \qquad {\prec_A} \subseteq \Point_A \times \Point_A \\[3pt]
    {\ordW} & = & \mathsf{tc}\big({\prec_A} \cup \{(x, \pplus{x})\} \cup \{(x, \pplus{\dual{x}})\}\big) \qquad
    x \preccurlyeq_W y \iff x = y \lor x \ordW y \\[3pt]
    \cons\,W & \iff & \forall x.\ x \not\ordW x \\[3pt]
    W \wext W' & \iff & A \subseteq A' \land {\ordW} \subseteq {\ordW[W']} \\[3pt]
    W_1 \wjoin W_2 & = & (A_1 \cup A_2,\ {\prec_{A_1}} \cup {\prec_{A_2}}) \\[3pt]
    W \grow{\rho}{C} & = & (A \cup \{\rho\},\ {\prec_A} \cup C[\rho])
  \end{array}
  \]
  \figrule
  \Description{Grammar of points, the moves of an end, and the definitions of worlds, their order, consistency, extension, growth and join.}
  \caption{Points over constant bases $\rho$ and point variables $\iota$, moves and worlds. An action moves its end from $x$ to $\pplus{x}$, a choice to $\mathsf{sel}$ or $\mathsf{bra}$, the payload ends of a message at $x$ arrive at $\mathsf{payload}(x, n)$, $A$ is the set of bases of $W$, $\Point_A$ the set of points with a base in $A$ and $\mathsf{tc}$ the transitive closure (restriction and the join lemma in \cref{app:w-ext}).}
  \label{fig:worlds}
\end{figure}
