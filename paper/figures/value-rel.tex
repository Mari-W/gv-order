% The value relation, one rule per type constructor. From map block v-rel,
% deadlock-free reading (beta < alpha). The lock-free index arithmetic
% beta + kappa <= alpha is in the appendix (PLAN.md, simplifications).
\begin{figure}[tbp]
  \small
  \judghead{Values}{v \in \Vrel{W}{\alpha}{\Theta}{T}}
  \begin{rules}
    \infer*[lab=V-Unit]{ }{\ast \in \Vrel{W}{\alpha}{\varnothing}{\ast}}
    \label{rule:V-Unit}
    \and
    \infer*[lab=V-Sess]{S' \approx S}{o \in \Vrel{W}{\alpha}{\{o \mapsto S'_x\}}{S_x}}
    \label{rule:V-Sess}
    \and
    \infer*[lab=V-Prod]
      {v_1 \in \Vrel{W}{\alpha}{\Theta_1}{T_1} \\ v_2 \in \Vrel{W}{\alpha}{\Theta_2}{T_2}}
      {\tpair{v_1}{v_2} \in \Vrel{W}{\alpha}{\Theta_1 \uplus \Theta_2}{T_1 \otimes T_2}}
    \label{rule:V-Prod}
    \and
    \infer*[lab=V-Sum]
      {v \in \Vrel{W}{\alpha}{\Theta}{T_i}}
      {\tinj_i\,v \in \Vrel{W}{\alpha}{\Theta}{T_1 \oplus T_2}}
    \label{rule:V-Sum}
  \end{rules}
  \begin{rules}
    \infer*[lab=V-Arrow]
      {\mathrm{dom}\,\Theta = \fends(e) \\ l \preccurlyeq_W \ap(\Theta) \\\\
       \forall W' \sqsupseteq W,\ \beta < \alpha,\ F,\ \Theta',\ w.\ \
       W' \vdash u \prec F \land w \in \Vrel{W'}{\beta}{\Theta'}{T}
       \ \Rightarrow\ e[w/b] \in \Erel{W'}{\beta}{\Theta \cup \Theta'}{U}{F}}
      {\tlam{b}{e} \in \Vrel{W}{\alpha}{\Theta}{T \arr{u}{l} U}}
    \label{rule:V-Arrow}
    \and
    \infer*[lab=V-Box]
      {\fends(e) = \varnothing \\\\
       \forall W' \sqsupseteq W,\ \bar x,\ \bar s \in \Inst,\ \beta < \alpha,\ F.\ \
       W' \vdash C[\bar x] \land W' \vdash u[\bar x] \prec F
       \ \Rightarrow\ e[\tbox{f}{e}/f] \in \Erel{W'}{\beta}{\varnothing}{U[\bar s][\bar x]}{F}}
      {\tbox{f}{e} \in \Vrel{W}{\alpha}{\varnothing}{\Box(\forall\bar\iota\,\bar Z.\,C.\ \ast \multimap^{u} U)}}
    \label{rule:V-Box}
    \and
    \infer*[lab=V-Ex]
      {l \preccurlyeq_W \ap(\Theta) \\ W \vdash C[\bar x] \\ v \in \Vrel{W}{\alpha}{\Theta}{T[\bar x]}}
      {v \in \Vrel{W}{\alpha}{\Theta}{\exists^{l}\bar\iota.\,C.\,T}}
    \label{rule:V-Ex}
  \end{rules}
  \figrule
  \Description{Inference rules for the value relation, one per type constructor.}
  \caption{The value relation, by recursion on $\alpha$ and then on $T$, under \cref{def:rel-anatomy}. Each rule is one clause, and $W' \sqsupseteq W$ ranges over the larger worlds of \cref{fig:worlds}.}
  \label{fig:value-rel}
\end{figure}
