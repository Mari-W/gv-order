% Selected typing rules. From map blocks r-fun (E-App), r-data
% (E-Case), r-act (E-Send, E-Close, E-Link). Contractivity premises are
% left to the convention of sec:types (PLAN.md, presentation).
\begin{figure}[tbp]
  \small
  \judghead{Typing terms}{\cGamma ; \lenv \vdash^{F}_{\lf{N}} \tm : \ty \eff{\varepsilon}}
  \begin{rules}
    \infer*[lab=E-App]
      {\tyj{\cGamma}{\lenv_1}{F \held \lenv_2}{\tm_1}{(\ty \arr{u}{l}\lf{[\varepsilon]}\, \tyB)\lf{^{a}}}\eff{\varepsilon_1} \\\\
       \tyj{\cGamma}{\lenv_2\lf{-\varepsilon_1}}{F \held (\ty \arr{u}{l} \tyB)\lf{^{a}}}{\tm_2}{\ty}\eff{\varepsilon_2} \\ u \prec F \\ \lf{\varepsilon_2 \leq a}}
      {\tyj{\cGamma}{\lenv_1, \lenv_2}{F}{\tm_1\,\tm_2}{\tyB}\eff{\varepsilon_1 + \varepsilon_2 + \varepsilon}}
    \label{rule:E-App}
    \and
    \infer*[lab=E-Case]
      {\tyj{\cGamma}{\lenv_1}{F \held \lenv_2}{\tm}{\ty_1 \tyoplus \ty_2}\eff{\varepsilon_0} \\
       \tyj{\cGamma}{\lenv_2\lf{-\varepsilon_0}, \vr{} : \ty_i}{F}{\tm_i}{\ty}\eff{\varepsilon_i}\ \ (i \in \{1, 2\})}
      {\tyj{\cGamma}{\lenv_1, \lenv_2}{F}{\tcasee{\tm}{\vr{}}{\tm_1}{\vr{}}{\tm_2}}{\ty}\eff{\varepsilon_0 + \max(\varepsilon_1, \varepsilon_2)}}
    \label{rule:E-Case}
  \end{rules}
  \begin{rules}
    \infer*[lab=E-Send]
      {\unfold(\sty) = \sSend{\bar\iota}{\ty}.\sty' \\ x \prec \payload{\ty}{x} \\ x \prec F \\\\
       \tyj{\cGamma}{\lenv}{F}{\tm}{\ty' \tyotimes \sty_x\lf{^{a,m}}}\eff{\varepsilon} \\ \ty'\lf{-1} \leq \payload{\ty}{x}}
      {\tyj{\cGamma}{\lenv}{F}{\tsend\,\tm}{\sty'_{\pplus{x}}\lf{^{m,m}}}\eff{\varepsilon + 1}}
    \label{rule:E-Send}
    \and
    \infer*[lab=E-Close]
      {\unfold(\sty) = \Close \\ x \prec F \\\\ \tyj{\cGamma}{\lenv}{F}{\tm}{\sty_x\lf{^{a,m}}}\eff{\varepsilon}}
      {\tyj{\cGamma}{\lenv}{F}{\tclose\,\tm}{\tyunit}\eff{\varepsilon + 1}}
    \label{rule:E-Close}
    \and
    \infer*[lab=E-Link]
      {\tyj{\cGamma}{\lenv}{F}{\tm}{\sty_x\lf{^{a_1,m_1}} \tyotimes \dual{\sty}_{\dual{x}}\lf{^{a_2,m_2}}}\eff{\varepsilon} \\\\
       \lf{a_1 + m_2 > N} \\ \lf{a_2 + m_1 > N}}
      {\tyj{\cGamma}{\lenv}{F}{\tlink\,\tm}{\tyunit}\eff{\varepsilon}}
    \label{rule:E-Link}
  \end{rules}

  \label{fig:typing-core}
\end{figure}
