  \judghead{Creation, boxes and existentials}{\cGamma ; \lenv \vdash^{F}_{\lf{N}} \tm : \ty \eff{\varepsilon}}
  \begin{rules}
    \infer*[lab=E-New]
      {\unl\,\lenv \\ \acyc_{\rho}(C) \\ \conserv_{\cGamma}(\rho, C) \\\\
       \lf{N \in \mathbb{N} \Rightarrow \rankable_{\cGamma}(\rho, C)} \\ \lf{m_1 + m_2 \leq N}}
      {\tyj{\cGamma}{\lenv}{F}{\tnew}{\tyexists^{\bot}\rho.\ C.\ \sty_{\rho}\lf{^{m_1,m_1}} \tyotimes \dual{\sty}_{\dual{\rho}}\lf{^{m_2,m_2}}}\eff{0}}
    \label{rule:E-New}
    \and
    \infer*[lab=E-New-Field]
      {\unl\,\lenv \\ \lf{m_1 + m_2 \leq N}}
      {\tyj{\cGamma}{\lenv}{F}{\tnew}{\sty_{\fld{x}{\kappa}}\lf{^{m_1,m_1}} \tyotimes \dual{\sty}_{\dual{\fld{x}{\kappa}}}\lf{^{m_2,m_2}}}\eff{0}}
    \label{rule:E-New-Field}
  \end{rules}
  \begin{rules}
    \infer*[lab=E-Box]
      {B = \tyBox(\forall \bar\iota\,\bar Z.\ C.\ \ast \tymultimap^{u}\lf{[\varepsilon]}\,\tyB) \\ \unl\,\lenv \\\\
       \forall \bar s \in \Inst.\ \tyj{\cGamma, C(\bar\iota)}{\lenv, f : B}{\up{u}}{\tm}{\tyB[\bar s]}\eff{\varepsilon}}
      {\tyj{\cGamma}{\lenv}{F}{\tbox{f}{\tm}}{B}\eff{0}}
    \label{rule:E-Box}
    \and
    \infer*[lab=E-Force]
      {\tyj{\cGamma}{\lenv}{F}{\tm}{\tyBox(\forall \bar\iota\,\bar Z.\ C.\ \ast \tymultimap^{u}\lf{[\varepsilon]}\,\tyB)}\eff{\varepsilon_0} \\\\
       \cGamma \vdash C[\bar x] \\ u[\bar x] \prec F}
      {\tyj{\cGamma}{\lenv}{F}{\tforce\,\tm}{\tyB[\bar s][\bar x]}\eff{\varepsilon_0 + 1 + \varepsilon}}
    \label{rule:E-Force}
    \and
    \infer*[lab=E-Open]
      {\lenv(a) = \tyexists^{l}\bar\iota.\ C.\ \ty \\ \neg\,\unl\,\ty \\ \conserv_{\cGamma}(\bar\iota, C) \\\\
       \tyj{\cGamma, C(\bar\iota)}{\lenv[a \mapsto \ty]}{F}{\tm}{\tyB}\eff{\varepsilon}}
      {\tyj{\cGamma}{\lenv}{F}{\tm}{\tyB}\eff{\varepsilon}}
    \label{rule:E-Open}
  \end{rules}
