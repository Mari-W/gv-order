  (** The side conditions of a fusion of a component of a cut. *)
  Local Lemma conn_fside1 Σ1 ΣC Σ2 ΣD (p1 p2 : proc) D U : tfree U →
    conn Σ1 ΣC p1.1 Σ2 ΣD p2.1 → fside D (Σ1 ∪ ΣC) U p1.
  Proof.
    intros HU Hc. destruct Hc as [? ? ? Hf1 ? HfC ? H1C ? ? HC1 ?]. split_and!; [|done|].
    - rewrite chans_of_union. set_solver.
    - by apply pfree_union.
  Qed.

  Local Lemma conn_fside2 Σ1 ΣC Σ2 ΣD (p1 p2 : proc) D U : tfree U →
    conn Σ1 ΣC p1.1 Σ2 ΣD p2.1 → fside D (Σ2 ∪ ΣD) U p2.
  Proof. intros HU Hc. apply (conn_fside1 Σ2 ΣD Σ1 ΣC p2 p1 D U HU). by apply conn_sym. Qed.

  (** The clauses of the composite. *)
  Local Lemma step W Σ1 ΣC Σ2 ΣD p1 p2 :
    PB1 W (Σ1 ∪ ΣC) p1 →
    Pbody κ n (Vrel κ) (Prel κ) (Vrel κ n) X2 W (Σ2 ∪ ΣD) TUnit p2 →
    conn Σ1 ΣC p1.1 Σ2 ΣD p2.1 → Prel κ n W (Σ1 ∪ Σ2) T (pconn p1 p2 (chans_of ΣC)).
  Proof.
    intros B1 B2 Hc. apply Prel_iff. intros _ W1 HW1.
    pose proof (Pbody_P _ _ _ _ _ B1) as P1. pose proof (Pbody_P _ _ _ _ _ B2) as P2.
    assert (tfree TUnit) as HTU by done.
    assert (∀ W Σ T p, X1 W Σ T p → Prel κ n W Σ T p) as X1P by (by intros ? ? ? ? [? _]).
    assert (∀ W Σ T p, X2 W Σ T p → Prel κ n W Σ T p) as X2P by (by intros ? ? ? ? [? _]).
    assert (∀ j W', olt j n → ConnL κ j W' T) as L by (intros j W' Hj; by apply IH).
    assert (∀ h D, fuse_ok h D (Σ1 ∪ ΣC) p1 →
      Prel κ n W1 (sdrop D (kren h (Σ1 ∪ ΣC))) T (pfuse h D p1)) as HP1.
    { intros h D Hh. apply Prel_fuse; [done|by eapply conn_fside1|]. by eapply Prel_mono. }
    assert (∀ h D, fuse_ok h D (Σ2 ∪ ΣD) p2 →
      Prel κ n W1 (sdrop D (kren h (Σ2 ∪ ΣD))) TUnit (pfuse h D p2)) as HP2.
    { intros h D Hh. apply Prel_fuse; [done|by eapply conn_fside2|]. by eapply Prel_mono. }
    assert (∀ W' p1' Σa h D Σ1' ΣC' Σ2' ΣD' p2', W1 ⊑ W' → X1 W' Σa T p1' → fuse_ok h D Σa p1' → pfree Σa →
      sdrop D (kren h Σa) = Σ1' ∪ ΣC' → Prel κ n W' (Σ2' ∪ ΣD') TUnit p2' →
      conn Σ1' ΣC' (pfuse h D p1').1 Σ2' ΣD' p2'.1 →
      Prel κ n W' (Σ1' ∪ Σ2') T (pconn (pfuse h D p1') p2' (chans_of ΣC'))) as G1.
    { intros W' p1' Σa h D Σ1' ΣC' Σ2' ΣD' p2' _ [_ HΦ] Hh Hpf EΣ HP Hcc.
      refine (HΦ h D (pfuse h D p1') (Σ1' ∪ ΣC') _ (gfuse_pfuse h D Σa p1' Hh) _ eq_refl Σ1' ΣC' Σ2' ΣD' p2' eq_refl HP Hcc).
      - split_and!; [|done|done]. by apply (fuse_side_chans h D Σa p1' Σ1' ΣC' Σ2' ΣD' p2'.1).
      - rewrite EΣ. apply senv_eqv_refl. }
    assert (∀ W' p2' Σb h D Σ1' ΣC' Σ2' ΣD' h1 D1, W1 ⊑ W' → X2 W' Σb TUnit p2' → fuse_ok h D Σb p2' → pfree Σb →
      sdrop D (kren h Σb) = Σ2' ∪ ΣD' → fuse_ok h1 D1 (Σ1 ∪ ΣC) p1 →
      sdrop D1 (kren h1 (Σ1 ∪ ΣC)) = Σ1' ∪ ΣC' →
      conn Σ1' ΣC' (pfuse h1 D1 p1).1 Σ2' ΣD' (pfuse h D p2').1 →
      Prel κ n W' (Σ1' ∪ Σ2') T (pconn (pfuse h1 D1 p1) (pfuse h D p2') (chans_of ΣC'))) as G2.
    { intros W' p2' Σb h D Σ1' ΣC' Σ2' ΣD' h1 D1 HW' [_ HΦ] Hh Hpf EΣ Hh1 EΣ1 Hcc.
      refine (HΦ h D (pfuse h D p2') (Σ2' ∪ ΣD') _ (gfuse_pfuse h D Σb p2' Hh) _ eq_refl Σ1' ΣC' Σ2' ΣD' _ eq_refl _ Hcc).
      - split_and!; [|done|done]. apply (fuse_side_chans h D Σb p2' Σ2' ΣD' Σ1' ΣC' (pfuse h1 D1 p1).1); [done|done|].
        by apply conn_sym.
      - rewrite EΣ. apply senv_eqv_refl.
      - rewrite <-EΣ1. apply Pbody_gfuse; [done|apply Φfuse_closed|by intros ? ? ? ? [? _]|by eapply conn_fside1|by apply gfuse_pfuse|].
        eapply Pbody_later; [|exact B1]. by eapply wext_trans. }
    exact (pc_clauses _ _ _ _ _ _ _ _ _ _ _ _
      (CCtx κ n W1 T X1 X2 Σ1 ΣC Σ2 ΣD p1 p2 Hn HT Hc (B1 W1 HW1) (B2 W1 HW1) L X1P X2P HP1 HP2 G1 G2)).
  Qed.
